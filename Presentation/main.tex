\section{Introducci\'on y Objetivos}
% --- Lámina 1: Contexto del Negocio ---
\begin{frame}{Introducción: Contexto del Negocio}
    \begin{itemize}
        \item <1-> \textbf{Decisión de Inversión:} En la industria musical, predecir si un track será un éxito es vital para determinar el destino de los recursos.
        \vspace{0.8em}
        \item <1-> \textbf{Plataformas Digitales:} Spotify genera millones de puntos de datos diarios sobre el comportamiento del usuario.
        \vspace{0.8em}
        \item <1-> \textbf{Competitividad:} Con miles de lanzamientos diarios, destacar un track requiere más que solo intuición artística.
    \end{itemize}
    \vspace{1em}
    \begin{block}{Oportunidad}
        El uso de \textbf{Minería de Datos} permite transformar características musicales en ventajas competitivas para sellos y artistas.
    \end{block}
\end{frame}
% --- Lámina 2: Definición del Problema ---
\begin{frame}{Definición del Problema}
    \begin{description}
        \item[Desafío] Predecir el éxito (popularidad) de una canción antes o durante su lanzamiento. 
        \item[Incertidumbre] Las inversiones en marketing musical suelen ser de alto riesgo sin un respaldo analítico de los datos.
    \end{description}
    
    \vspace{1em}
    \textbf{Pregunta de Investigación:}
    \vspace{0.5em}
    \par¿Es posible determinar el \textit{ranking} de una canción basándose en sus atributos técnicos como el tempo, la energía y la bailabilidad?
\end{frame}
%----- Lámina 3: Objetivos ---
\begin{frame}{Objetivos}
    \textbf{Objetivo General:}
    \vspace{0.5em}
    \begin{itemize}
        \item Aplicar el proceso de minería de datos para predecir el ranking de tracks de Spotify utilizando la metodología CRISP-DM.
    \end{itemize}
    \vspace{1em}
    \textbf{Objetivos Específicos:}
    \vspace{0.5em}
    \begin{itemize}
        \item Ejecutar el preprocesamiento y limpieza del dataset.
        \item Realizar un análisis exploratorio de datos (EDA) para identificar patrones y relaciones entre variables.
        \item Desarrollar un modelo predictivo para pronosticar el ranking de una canción.
    \end{itemize}
\end{frame}
%----- Sección de Metodología ---
\section{Metodolog\'ia CRISP-DM}
%----- Lámina 4: Características de CRISP-DM ---
\begin{frame}{Metodología CRISP-DM: Características}
    \begin{columns}[c]
        \column{.5\textwidth}
        \begin{itemize}
            \item <1-> \textbf{Estándar de la Industria:} Es el modelo de proceso más utilizado para proyectos de Minería de Datos.
            \vspace{0.8em}
            \item <2-> \textbf{Ciclo Iterativo:} Permite retroceder entre fases para garantizar la calidad del modelo final.
            \vspace{0.8em}
            \item <3-> \textbf{Enfoque en el Negocio:} No solo se centra en el algoritmo, sino en resolver un problema real.
        \end{itemize}
        
        \column{.45\textwidth}
        \begin{figure}
            \centering
            % Asegúrate de tener la imagen en tu carpeta de archivos
            \includegraphics[width=\textwidth]{figs/CRISP-DM.jpg}
            \caption{\footnotesize Ciclo de vida CRISP-DM.}
        \end{figure}
    \end{columns}
\end{frame}
%----- Lámina 5: Justificación de la Metodología ---
\begin{frame}{Justificación de la Metodología}
    \begin{block}{Motivo de elección}
        Se selecciona esta metodología por su capacidad para estructurar un proceso completo desde la comprensión del negocio y de los datos técnicos hasta la predicción de resultados.
    \end{block}
    \vspace{0.5cm}
    \textbf{Ventajas de su utilización:}
    \vspace{0.8em}
    \begin{enumerate}
        \item \textbf{Flexibilidad:} Ideal para ajustar el preprocesamiento de variables acústicas según los hallazgos del EDA.
        \item \textbf{Evaluación Continua:} Permite validar si las métricas de regresión responden correctamente a la meta de negocio (predecir el ranking).
        \item \textbf{Estructura:} Facilita la transición entre el análisis exploratorio y el modelado de regresión propuesto.
    \end{enumerate}
\end{frame}
%----- Sección de Comprensión del Negocio y los Datos ---
\section[Comprensi\'on del Negocio y los Datos]{Comprensi\'on del Negocio y los Datos}
%----- Lámina 6: Comprensión del Negocio ---
\begin{frame}{Fase I: Comprensión del Negocio}
    \begin{block}{Objetivo de Minería de Datos}
        Transformar los atributos acústicos de los tracks en un valor numérico de \texttt{popularity} mediante un modelo de \textbf{regresión}, permitiendo predecir el rendimiento comercial de una canción.
    \end{block}
    
    \vspace{0.2cm}
    \textbf{Criterios de Éxito del Proyecto:}
    \begin{itemize}
        \item \textbf{Técnico:} Lograr un modelo con un margen de error aceptable que permita diferenciar canciones con alto potencial de aquellas con bajo impacto.
        \item \textbf{De Negocio:} Identificar los patrones históricos que correlacionan el "sonido" (variables flotantes) con el éxito, para reducir el riesgo en la toma de decisiones de inversión.
    \end{itemize}
    
    \vspace{0.2cm}
    \textbf{Meta Final:}
    \par Generar una herramienta analítica que determine si una canción será un éxito basándose en su composición técnica.
\end{frame}

\section[Preprocesamiento de Datos]{Preprocesamiento de Datos}
%----- Lámina 9: Preprocesamiento ---
\begin{frame}{Preprocesamiento: Valores Faltantes}
\begin{columns}
    % Imagen de vacíos
    \column{0.35\textwidth}
    \begin{figure}
        \centering
        \includegraphics[width=0.5\textwidth]{figs/nulls.png} 
        \vspace{-0.5em}
        \caption{\footnotesize Cantidad y porcentaje de valores faltantes por columna.}
    \end{figure}
    % Explicación
    \column{0.60\textwidth}
    \textbf{Tratamiento de Datos:}
    \vspace{0.5cm}
    \begin{itemize}
        \item <1-> \textbf{Limpieza Inicial:} Identificación de valores nulos en cada campo del dataset.
        \vspace{0.2cm}
        \item <2-> \textbf{Estrategia:} Dado el bajo porcentaje de nulos y su poca relevancia para el análisis, se opta por eliminar las filas con valores faltantes. Esto también permite la utilización de un modelo de regresión.
    \end{itemize}
\end{columns}
\end{frame}
%----- Lámina 10: Preprocesamiento ---
\begin{frame}{Preprocesamiento: Variables No Relevantes}
    \begin{itemize}
        \item <1-> \textbf{Limpieza de nulos:} Se ejecuta la eliminación de filas con valores faltantes.
        \vspace{0.2cm}
        \item <1-> \textbf{Limpieza de variables no relevantes:} Se descartan las columnas que no aportan información útil para el modelo predictivo, como los IDs, nombres y géneros de canciones.
    \end{itemize}
    \vspace{0.2cm}
    \par \setlength{\parindent}{1em}\textbf{Dimensiones resultantes:} 113.999 registros y 14 columnas.
    \begin{figure}
        \centering
        \includegraphics[width=0.9\textwidth]{figs/clean.png} 
        \vspace{-0.5em}
        \caption{\footnotesize Limpieza Estructural del Dataset.}
    \end{figure}
\end{frame}

\begin{frame}{Preprocesamiento: Valores Lógicos}
    \begin{itemize}
        \item <1-> \textbf{Valores Lógicos:} Los modelos de regresión necesitan variables numéricas. Por esta razón, se transforman los valores booleanos de la columna \texttt{explicit} a 0 (no explícito) y 1 (explícito).
